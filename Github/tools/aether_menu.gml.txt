
// ---------------------------------------------------------------------------------------------------------------------
// "Aether Tower": the pause-menu page for everything Rivals related. Built with PT's own menu framework (scr_menu), so it
// looks and behaves like the video / controls pages.
// ---------------------------------------------------------------------------------------------------------------------
#macro RIVALS_PAGE_ROOT 100
#macro RIVALS_PAGE_CHARS 101   // 101.. = character pages
#macro PRE_CSS_STATE 104
#macro RIVALS_CHARS_PER_PAGE 6

function rivals_register_lang()
{
	if !variable_global_exists("lang_map")
		return;
	var en = ds_map_find_value(global.lang_map, "en");
	if is_undefined(en)
		return;
	ds_map_set(en, "pause_aether", "AETHER TOWER");
	ds_map_set(en, "aether_enabled", "RIVALS");
	ds_map_set(en, "aether_character", "CHANGE CHARACTER");
	ds_map_set(en, "aether_reset", "RESET POSITION");
	ds_map_set(en, "aether_speed", "2X SPEED");
	ds_map_set(en, "aether_bounds", "SHOW BOUNDS");
	ds_map_set(en, "aether_recenter", "KEEP IN STAGE");
	ds_map_set(en, "aether_input", "INPUT");
	ds_map_set(en, "aether_workshop", "WORKSHOP / SKINS");
	ds_map_set(en, "aether_more", "MORE >>");
}

function rivals_set_speed(mult)
{
	var R = global.rivals;
	R.speed = mult;
	if R.connected
		external_call(R.fn.cmd, 6, 1, mult);
}

function rivals_open_css()
{
	var R = global.rivals;
	if !R.connected
		exit;
	R.css = true;
	R.ready = false;
	R.status = "pick a character in the Rivals window (Practice to confirm)";
	external_call(R.fn.cmd, 7, 0, 0);
}

function rivals_request_reset()
{
	global.rivals.pending_reset = true;
}

function rivals_menu_pick(idx)
{
	var R = global.rivals;
	if !R.enabled
		rivals_toggle();
	R.css = false;
	rivals_select(idx);
}

// builds the Aether Tower pages inside obj_option (called at the end of its Create event)
function rivals_option_menus()
{
	var R = global.rivals;
	if !R.available
		exit;
	var root = create_menu_fixed(RIVALS_PAGE_ROOT, menu_anchor.left, 150, 40, 0);
	root.exit_on_back = true;
	add_option_press(root, 0, "option_back", function()
	{
		instance_destroy();
	});
	add_option_toggle(root, 1, "aether_enabled", function(val)
	{
		if (global.rivals.enabled != val)
			rivals_toggle();
	}).value = R.enabled;
	add_option_press(root, 2, "aether_character", function()
	{
		menu_goto(RIVALS_PAGE_CHARS);
	});
	add_option_press(root, 3, "aether_reset", function()
	{
		rivals_request_reset();
		instance_destroy();
	});
	add_option_toggle(root, 4, "aether_speed", function(val)
	{
		rivals_set_speed(val ? 2 : 1);
	}).value = (R.speed > 1);
	add_option_toggle(root, 5, "aether_bounds", function(val)
	{
		global.rivals.show_bounds = val;
	}).value = R.show_bounds;
	add_option_toggle(root, 6, "aether_recenter", function(val)
	{
		global.rivals.recenter = val;
	}).value = R.recenter;
	add_option_multiple(root, 7, "aether_input", [create_option_value("AUTO", 0, false), create_option_value("RIVALS PAD", 1, false), create_option_value("PT KEYS", 2, false)], function(val)
	{
		global.rivals.input_mode = val;
	}).value = R.input_mode;
	array_push(menus, root);
	// character pages: the stock cast, a few per page
	var stock = [];
	for (var i = 0; i < array_length(R.catalog); i++)
	{
		if (R.catalog[i].source == "stock")
			array_push(stock, i);
	}
	var pages = max(1, ceil(array_length(stock) / RIVALS_CHARS_PER_PAGE));
	for (var pg = 0; pg < pages; pg++)
	{
		var m = create_menu_fixed(RIVALS_PAGE_CHARS + pg, menu_anchor.left, 150, 40, RIVALS_PAGE_ROOT);
		var ob = add_option_press(m, 0, "option_back", -4);
		ob.func = method(m, function()
		{
			with (obj_option)
				menu_goto(RIVALS_PAGE_ROOT);
		});
		if (pg == 0)
		{
			add_option_press(m, 1, "aether_workshop", function()
			{
				rivals_open_css();
				instance_destroy();
			});
		}
		for (var k = pg * RIVALS_CHARS_PER_PAGE; k < min(array_length(stock), (pg + 1) * RIVALS_CHARS_PER_PAGE); k++)
		{
			var c = R.catalog[stock[k]];
			var o = add_option_press(m, 2 + k, c.name, -4);
			o.localization = false;
			o.cat_index = stock[k];
			o.func = method(o, function()
			{
				rivals_menu_pick(cat_index);
				with (obj_option)
					menu_goto(RIVALS_PAGE_ROOT);
			});
		}
		if pages > 1
		{
			var mo = add_option_press(m, 100, "aether_more", -4);
			mo.next_page = RIVALS_PAGE_CHARS + ((pg + 1) mod pages);
			mo.func = method(mo, function()
			{
				var _np = next_page;
				with (obj_option)
					menu_goto(_np);
			});
		}
		array_push(menus, m);
	}
}

function rivals_pause_aether()
{
	fmod_event_one_shot("event:/sfx/ui/select");
	with (instance_create(x, y, obj_option))
	{
		depth = other.depth - 1;
		if (variable_global_exists("rivals") && global.rivals.available)
			menu_goto(RIVALS_PAGE_ROOT);
	}
}
